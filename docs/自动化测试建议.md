## 一、什么该自动化，什么不该

判断标准只有一条：**能不能给出确定的期望值**。

| 分层 | 内容 | 是否自动化 | 理由 |
|---|---|---|---|
| **L1 纯函数** | `parser.parse_tasks` / `find_deadline_date` / `score_task` / `clean_title` / `extract_name` | ✅ **首选** | 输入字符串 → 输出结构，无副作用、无时间依赖（可注入今天） |
| **L2 数据逻辑** | `Store.sync_history` / `purge_history` / 跨天归档 / `QuoteDeck` 发牌 / `i18n` 渲染 / 主题对比度 | ✅ **强烈建议** | 纯 JSON 读写，用 `tmp_path` 隔离即可，跑一次 < 1s |
| **L3 对话编排** | `NoteApp.respond()` 的多轮交互（追问/确认/撤回） | ✅ **建议**（需 tkinter） | 会创建真实窗口，但**不进 mainloop**，`update()` 推进即可；CI 用 `xvfb-run` |
| **L4 视觉呈现** | 字体渲染色差、卡片圆角、动画流畅度 | ⚠️ 半自动 | 只能截图后人工比对，或算像素覆盖率（成本高、收益低） |
| **L5 真人体验** | 「一句话说得顺不顺」「回复语气讨不讨喜」 | ❌ 不自动化 | 主观判断，靠 UAT 脚本 + 人工抽检 |

**结论**：把 L1+L2 做到 100% 覆盖（这是 bug 的高发区，且改动最频繁），L3 覆盖关键路径（追问/撤回/跨天），L4/L5 交给打包前的 UAT 脚本。

## 二、优先级排序（按「改了容易坏 × 坏了用户能感觉到」）

| 优先级 | 模块 | 为什么排这个位置 | 现有用例 |
|---|---|---|---|
| P0 | `parser` 解析器 | 每次改词表都可能连带影响别的句子，回归成本最高 | `test_parser.py` 69 条 |
| P0 | 历史归档 / 跨天 | 数据一旦记错日期无法补救，用户直接失去信任 | `test_history.py` 22 条 |
| P0 | 提醒去重 | 重复弹卡片是最刺眼的体验 bug | `test_core_logic.py::TestReminderDedup` |
| P1 | 哲言牌堆 | 「切语言串句子」是用户点名的关切 | `test_core_logic.py::TestQuotes` |
| P1 | i18n 完整性 | 漏译会让英文界面蹦中文 | `test_core_logic.py::TestI18n` |
| P1 | 进度条百分比 | 纯计算，边界多（0/满/奇偶） | `test_core_logic.py::TestProgressBar` |
| P2 | 撤回 / 追问多轮 | 交互状态机，靠 GUI 测试守 | `test_gui_flow.py` |
| P2 | 主题对比度 | 定量可测，防止加主题时翻车 | `test_core_logic.py::TestThemes` |

```python
    ("整理时间表", "整理时间表"),

    ("3号电池", "3号电池"),

    ("把水电费交了", "交水电费"),

    assert core.parse_tasks(text)[0][0] == title


def test_month_day_within_half_year_keeps_year():
    """半年内的月日保留今年（v1.9.4 规则）"""
    today = dt.date.today()
    d = core.find_deadline_date("%d月%d日交报告" % (today.month, today.day))
    assert d and d[0].year == today.year, d[0]


def test_month_day_beyond_half_year_rolls_next_year():
    """超过 184 天自动滚到明年"""
    today = dt.date.today()
    far = today - dt.timedelta(days=200)
    if abs((today - dt.date(today.year, far.month, far.day)).days) > 184:
        d = core.find_deadline_date("%d月%d日交报告" % (far.month, far.day))
        assert d and d[0].year == today.year + 1, d[0]


def test_multi_task_split():
    got = core.parse_tasks("9点开会，下午3点前交周报，顺便买瓶酱油")
    assert [x[0] for x in got] == ["开会", "交周报", "买瓶酱油"]
```

### 4.3 进度条百分比：把「取整口径」钉死

```python
# tests/test_core_logic.py
import pytest


def pct(done, total):
    """镜像 main.py 里的算法：int(round(ratio * 100))"""
    if total <= 0:
        return 0
    return int(round((done / total) * 100))


@pytest.mark.parametrize("done,total,expect", [
    (0, 5, 0), (1, 2, 50), (5, 5, 100),
    (1, 8, 12),      # 12.5 → 银行家舍入取偶 → 12
    (3, 8, 38),      # 37.5 → 取偶 → 38（注意与上一条方向相反！）
    (5, 8, 62),      # 62.5 → 取偶 → 62
    (7, 7, 100),
])
def test_progress_percent(done, total, expect):
    assert pct(done, total) == expect


def test_progress_never_exceeds_100():
    """脏数据保护：done > total 时不能显示 120%"""
    assert pct(7, 5) <= 100
```

> **这里暴露的是产品问题**：`1/8 → 12%`、`3/8 → 38%`，同一个分母一半向下取一半向上取。见《文档问题清单》D-07。测试把这些口径**固定下来**，将来谁改算法都会立刻失败，逼着人确认。

```python
import quotes, i18n


def _has_cjk(s):
    return any("\u4e00" <= ch <= "\u9fff" for ch in s)


def test_zh_deck_never_emits_english():
    deck = quotes.QuoteDeck()
    for _ in range(400):
        assert not _has_cjk(deck.draw()) is False   # 中文牌堆必须句句含汉字


def test_en_deck_never_emits_chinese():
    deck = quotes.QuoteDeck(quotes.QUOTES_EN)
    assert deck.quotes, "英文牌堆不能是空的"
    for _ in range(200):
        assert not _has_cjk(deck.draw())


def test_one_round_no_repeat():
    """一轮内绝不重复（文档八承诺）"""
    deck = quotes.QuoteDeck()
    n = len(deck.quotes)
    got = [deck.draw() for _ in range(n)]
    assert len(set(got)) == n


def test_cross_round_avoids_recent():
    """跨轮时最近 18 句排到末尾，避免刚说完又说一遍"""
    deck = quotes.QuoteDeck()
    n = len(deck.quotes)
    first = [deck.draw() for _ in range(n)]
    second = deck.draw()
    assert second not in first[-18:]
```

```python
# tests/test_core_logic.py

def should_fire(task, now_hhmm, today):
    """镜像 check_reminders 的判定顺序（不含 UI 副作用）"""
    if task.get("done"):
        return False
    if not task.get("hhmm") and not task.get("remind_at"):
        return False
    if not task.get("remind_before") and not task.get("remind_at"):
        return False
    if task.get("reminded"):
        return False
    if task.get("due_date") and task["due_date"] != today:
        return False
    return True


def test_fires_once(mk):
    t = mk("开会", hhmm="20:00", remind_before=15)
    today = "2026-10-30"
    assert should_fire(t, "19:45", today) is True
    t["reminded"] = True                       # 弹过之后置位
    assert should_fire(t, "19:46", today) is False


def test_done_task_never_fires(mk):
    assert should_fire(mk("开会", hhmm="20:00", remind_before=15, done=True),
                       "19:45", "2026-10-30") is False


def test_other_day_task_never_fires(mk):
    t = mk("开会", hhmm="20:00", remind_before=15, due_date="2026-10-31")
    assert should_fire(t, "19:45", "2026-10-30") is False


def test_no_remind_setting_never_fires(mk):
    assert should_fire(mk("开会", hhmm="20:00"), "19:45", "2026-10-30") is False
```

### 4.7 主题对比度：加新配色时的自动守门

```python
# tests/test_core_logic.py
import themes


def _lum(hex_color):
    c = hex_color.lstrip("#")
    r, g, b = [int(c[i:i + 2], 16) / 255 for i in (0, 2, 4)]
    f = lambda v: v / 12.92 if v <= 0.03928 else ((v + 0.055) / 1.055) ** 2.4
    return 0.2126 * f(r) + 0.7152 * f(g) + 0.0722 * f(b)


def _ratio(a, b):
    la, lb = _lum(a), _lum(b)
    if la < lb:
        la, lb = lb, la
    return (la + 0.05) / (lb + 0.05)


def test_all_themes_pass_aaa():
    """文档写「10:1 左右」，这里用 9.5 作为硬底线（鼠尾草绿实测 9.89）"""
    bad = []
    for tid in themes.ids():
        th = themes.get(tid)
        r = _ratio(th["bg"], th["text"])
        if r < 9.5:
            bad.append((tid, round(r, 2)))
    assert not bad, bad


def test_eight_themes_and_unique_ids():
    assert len(themes.ids()) == 8
    assert len(set(themes.ids())) == 8
```

### 4.8 GUI 对话流程（需 tkinter）

```python
# tests/test_gui_flow.py
import pytest
tk = pytest.importorskip("tkinter")
import main as M


@pytest.fixture
def app(tmp_path, monkeypatch):
    monkeypatch.setattr(M, "data_dir", lambda: str(tmp_path))
    root = tk.Tk()
    a = M.NoteApp(root)
    root.update()
    a.store.setup_stage = 3
    a.store.user_name, a.store.bot_name = "小李", "小记"
    a.store.tasks, a.store.messages = [], []
    a._ring_bell = lambda *x: None
    a._cards = []
    a.show_due_card = lambda t: a._cards.append(t.get("title"))
    yield a
    root.destroy()


def say(app, text, max_confirm=6):
    """发一句话；碰上过期确认就按用户习惯点确认（可能连续多条）"""
    r = app.respond(text)
    n = 0
    while isinstance(r, str) and "过截止日期" in r and n < max_confirm:
        r = app.respond("确认")
        n += 1
    return r


def test_undo_removes_just_added_task(app):
    """v1.9.9 修的真 bug：刚建的任务说「撤回」，
    旧代码会回「已撤掉截止时间」而任务仍在"""
    say(app, "买牛奶")
    assert len(app.store.tasks) == 1
    r = say(app, "撤回")
    assert app.store.tasks == []
    assert "撤掉" in r


def test_undo_without_deadline_does_not_lie(app):
    """任务本来没设时间，不能谎称撤掉了时间"""
    say(app, "开会")
    app._undo_last = {"kind": "deadline", "id": app.store.tasks[0]["id"]}
    r = app.respond("撤一下时间")
    assert "本来就没设" in r
    assert len(app.store.tasks) == 1          # 任务不该被删


def test_expired_reply_letter_does_not_loop(app):
    """v1.9.9：回字母 A 曾导致无限追问，最后把「确认」记成任务名"""
    say(app, "数学作业")
    app.respond("A，凌晨三点")                # 一定已过 → 进过期确认
    r = app.respond("A")                      # 用字母确认，不能再问一次
    assert "过截止日期" not in r
    assert all(t["title"] != "确认" for t in app.store.tasks)


def test_expired_multi_queue_all_handled(app):
    """v1.9.9：一句话里两个过期任务，曾经只处理第一个，其余静默丢弃"""
    say(app, "凌晨2点写方案，凌晨3点看球")
    titles = [t["title"] for t in app.store.tasks]
    assert "写方案" in titles and "看球" in titles
```

| 闸门 | 内容 | 触发时机 | 时长 |

| ① 快闸 | `test_parser.py` + `test_history.py` | 每次改 `parser.py`/`main.py` | < 1s |

| ③ 验收闸 | `python main.py` 手动走一遍《测试用例设计》里的 P1 用例 | 打包发版前 | 10min |

## 六、还没自动化、建议后续补的

| 缺口 | 难点 | 建议做法 |
|---|---|---|
| 卡片叠放位置 | 需要真实窗口坐标 | 断言 `DueCard.slot` 递增即可（不比像素） |
| 跨天「一直开着」路径 | 要等 60s 轮询 | 把轮询体抽成 `app._check_rollover()` 再单测 |
| 时钟 24h 不停 | 真实时间太长 | 用假 `time.time` 快进 86400 次心跳，断言最终显示值 |
| 休眠唤醒 | 无法模拟 | 只能人工验证，写进 UAT 清单 |
| 英文引擎的中文兜底 | 现在只测了少量句子 | 把中文 69 条用例批量喂给 `core_en`，跑「双引擎对照」 |
